```python
# dashboard.py - Nova função para detectar mudanças
def _get_csv_hash():
    stat = os.stat(CSV_PATH_ALT)
    return f"alt_{stat.st_mtime}"

# Decorator com TTL + validação
@st.cache_data(hash_funcs={_get_csv_hash: str}, ttl=300)
def carregar_dados():
    _ = _get_csv_hash()  # Força revalidação
    # ...
```
